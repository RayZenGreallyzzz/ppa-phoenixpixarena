function playerAnimDef(name){

    const a=GNOME_ANIM[name];
    if(a.img&&a.img.complete&&a.img.naturalWidth>0)return a;
    return GNOME_ANIM.idle;
  }
  if(playerUsesArcherSprites() && ARCHER_ANIM[name]){
    const a=ARCHER_ANIM[name];
    if(a.img&&a.img.complete&&a.img.naturalWidth>0)return a;

  }
  if(playerUsesAssassinSprites() && ASSASSIN_ANIM[name]){

    if(a.img&&a.img.complete&&a.img.naturalWidth>0)return a;
    return ASSASSIN_ANIM.idle;
  }
  if(playerUsesTankSprites() && TANK_ANIM[name]){
    const a=TANK_ANIM[name];
    if(a.img&&a.img.complete&&a.img.naturalWidth>0)return a;
    return TANK_ANIM.idle;
  }
  if(playerUsesBerserkerSprites() && BERSERKER_ANIM[name]){
    const a=BERSERKER_ANIM[name];

    return BERSERKER_ANIM.idle;
  }
  if(playerUsesPriestSprites() && PRIEST_ANIM[name]){
    const a=PRIEST_ANIM[name];
    if(a.img&&a.img.complete&&a.img.naturalWidth>0)return a;
    return PRIEST_ANIM.idle;
  }
  if(playerUsesMageSprites() && MAGE_ANIM[name]){
    const a=MAGE_ANIM[name];

  return ANIM[name]||ANIM.idle;
}
// Canonical direction index: 0=N,1=NE,2=E,3=SE,4=S,5=SW,6=W,7=NW.
function dir8Canonical(dx,dy){
  if(Math.hypot(dx,dy)<0.001)return Number.isFinite(P.dir8)?P.dir8:4;
  let a=Math.atan2(dy,dx);
  let n=(a+Math.PI/2)%(Math.PI*2);
  if(n<0)n+=Math.PI*2;
  return Math.round(n/(Math.PI/4))%8;
}
function playerDir8(){
  // Run-and-attack: movement continues in joystick direction, but the
  // character visually turns toward the shot/target for the attack moment.
  const aimingNow=(P.attacking||P.runAttackT>0||P.shootT>0);
  if(aimingNow && Number.isFinite(P.meleeAng)){
    P.dir8=dir8Canonical(Math.cos(P.meleeAng),Math.sin(P.meleeAng));
    return P.dir8;
  }
  if(Math.hypot(P.vx,P.vy)>0.15){
    P.dir8=dir8Canonical(P.vx,P.vy);
  }
  if(!Number.isFinite(P.dir8))P.dir8=4;
  return P.dir8;
}

    // Animated cave spider: 4 walking frames, 4 directions.
    if(isCaveSpider && CAVE_SPIDER_ATLAS.complete && CAVE_SPIDER_ATLAS.naturalWidth>0){
      const sf=e.spiderMoving
        ? ((Math.floor((Date.now()+(e.spiderPhase||0)*55)/125))%4)
        : 0;
      const sr=(e.spiderDir===undefined?1:e.spiderDir);
      const dh=42*eliteScale; // elite is visual-only ×2
      const dw=72*eliteScale;
      cx.save();
      cx.imageSmoothingEnabled=false;
      if(e.flash>0){
        cx.globalAlpha=0.72;
        cx.filter='brightness(1.8)';
      }
      cx.drawImage(
        CAVE_SPIDER_ATLAS,
        sf*CAVE_SPIDER_FW,sr*CAVE_SPIDER_FH,CAVE_SPIDER_FW,CAVE_SPIDER_FH,
        sx-dw/2,sy-dh*0.72+bob,dw,dh
      );
      cx.filter='none';
      cx.globalAlpha=1;
      cx.restore();
    }
    // Approved animated mob sheets: 4 directions × 4 movement frames.
    else if(isApprovedAnimated && e.animPack.img.complete && e.animPack.img.naturalWidth>0){
      const af=e.animMoving
        ? ((Math.floor((Date.now()+(e.animPhase||0)*43)/e.animPack.ms))%4)
        : 0;
      const ar=(e.animDir===undefined?1:e.animDir);
      const dh=e.animPack.h*eliteScale;
      const dw=dh*(MOB_ANIM_FW/MOB_ANIM_FH);
      cx.save();
      cx.imageSmoothingEnabled=false;
      if(e.flash>0){
        cx.globalAlpha=0.72;
        cx.filter='brightness(1.8)';
      }
      cx.drawImage(
        e.animPack.img,
        af*MOB_ANIM_FW,ar*MOB_ANIM_FH,MOB_ANIM_FW,MOB_ANIM_FH,
        sx-dw/2,sy-dh*0.78+bob,dw,dh
      );
      cx.filter='none';
      cx.globalAlpha=1;
      cx.restore();
    }
    // Fart Zone guards: always draw the COMPLETE 128×128 source image.
    // A small foot-pivot step animation prevents static sprites from "floating".
    else if(e.isFartGuard && e.sprite && e.sprite.complete && e.sprite.naturalWidth>0){
      const baseH=Math.max(88,Math.min(112,e.sz*2.55))*eliteScale;
      const baseW=baseH*(e.sprite.naturalWidth/e.sprite.naturalHeight);
      const moving=!!e.fartMoving;
      const phase=(Date.now()/105)+(e.fartWalkPhase||0);
      const step=moving?Math.sin(phase):0;
      const lift=moving?Math.abs(Math.sin(phase))*2.2:0;
      const tilt=moving?step*0.035:0;
      const sxScale=moving?(1+Math.abs(step)*0.018):1;
      const syScale=moving?(1-Math.abs(step)*0.025):1;

      cx.save();
      cx.imageSmoothingEnabled=false;
      if(e.flash>0){cx.globalAlpha=.74;cx.filter='brightness(1.8)';}
      cx.translate(sx,sy+bob-lift);
      cx.rotate(tilt*(e.face||1));
      cx.scale((e.face||1)*sxScale,syScale);
      // Pivot from feet. No source cropping = no chopped heads/arms/weapons.
      cx.drawImage(e.sprite,-baseW/2,-baseH*.80,baseW,baseH);
      cx.filter='none';cx.globalAlpha=1;
      cx.restore();
    }
    // Other dungeon monster sprites.
    else if(e.sprite && e.sprite.complete && e.sprite.naturalWidth>0){
      // Scavenger slime is exactly about 1/4 of the current player on screen:
      // 22 world px × 1.18 camera = ~26 px. Other monsters keep their old sizes.
      const dh=(isScavengerSlime?25:Math.max(54,Math.min(92,e.sz*2.0)))*eliteScale;
      const widthMul=isScavengerSlime?(e.slimeWidthMul||1):1;
      const dw=dh*(e.sprite.naturalWidth/e.sprite.naturalHeight)*widthMul;
      cx.save();
      cx.imageSmoothingEnabled=false;
      // Tiny hit flash without replacing the sprite silhouette.
      if(e.flash>0){
        cx.globalAlpha=0.72;
        cx.filter='brightness(1.8)';
      }
      if(isScavengerSlime&&e.slimeFlip){
        cx.translate(sx,0);
        cx.scale(-1,1);
        cx.drawImage(e.sprite,-dw/2,sy-dh*0.78+bob,dw,dh);
      }else{
        cx.drawImage(e.sprite,sx-dw/2,sy-dh*0.78+bob,dw,dh);
      }
      cx.filter='none';
      cx.globalAlpha=1;
      cx.restore();
    }else{
      // Fallback only if a sprite failed to load.
      cx.fillStyle=e.flash>0?'#ffffff':e.col;
      cx.beginPath();cx.arc(sx,sy+bob,e.sz*0.42*eliteScale,0,Math.PI*2);cx.fill();
    }
    // HP bar
    if(e.hp<e.mhp||e.isDungeonElite){
      const bw=e.sz*(e.isDungeonElite?2.3:1),bh=e.isDungeonElite?7:4;
      const barY=sy-e.sz*0.65*eliteScale;
      cx.fillStyle='rgba(0,0,0,0.78)';
      cx.fillRect(sx-bw/2,barY,bw,bh);
      cx.fillStyle=e.isDungeonElite?'#b25cff':'#ff3333';
      cx.fillRect(sx-bw/2,barY,bw*Math.max(0,e.hp/e.mhp),bh);
      if(e.isDungeonElite){cx.strokeStyle='#f2c45e';cx.lineWidth=1;cx.strokeRect(sx-bw/2,barY,bw,bh)}
    }
    // Уровень и имя моба
    if(e.lvl&&e.type&&e.type.n){
      cx.textAlign='center';
      cx.font=e.isDungeonElite?'bold 11px monospace':'bold 9px monospace';
      const _eliteName=(e.isDungeonElite?'⚠ ЭЛИТА · ':'')+'['+e.lvl+'] '+e.type.n;
      const _nameY=sy-e.sz*0.75*eliteScale;
      cx.fillStyle='rgba(0,0,0,0.82)';
      cx.fillText(_eliteName,sx+1,_nameY+1);
      cx.fillStyle=e.isDungeonElite?'#ffd76a':'#ffdd88';
      cx.fillText(_eliteName,sx,_nameY);
      cx.textAlign='left';
    }
  }

  // Crystal projectiles from the world boss.
  if(P.scene==='worldboss')drawWorldCrystalProjectiles();

  // Boss fireballs
  for(const f of PR){
    const fx=f.x-cam.x,fy=f.y-cam.y;
    const fg=cx.createRadialGradient(fx,fy,0,fx,fy,15);
    fg.addColorStop(0,'rgba(255,240,180,0.95)');
    fg.addColorStop(0.4,'rgba(255,140,40,0.75)');
    fg.addColorStop(1,'rgba(255,60,0,0)');
    cx.fillStyle=fg;cx.beginPath();cx.arc(fx,fy,15,0,Math.PI*2);cx.fill();
  }

  // Online players from Realtime Presence (same scene only).
  drawOnlinePlayers();

  // Equipped visuals.
  updatePetFollower();
  drawPetFollower();

  // Wings and cloak stay as equipment/craft items,
  // but are NOT rendered in the world.
  // Player is always rendered in the normal world pass.
  // The old reward-only late-render branch could make the hero invisible
  // between Titan death and personal-chest pickup on some mobile browsers.
  drawPlayer();

  // Player projectiles are rendered above world sprites so shots stay readable.
  try{drawPlayerCannonballs()}catch(_){}
  try{drawPlayerArrows()}catch(_){}

  // Artifact floats above/beside the character.
  drawEquippedArtifact();

  // V104 depth for the original Phoenix and all ordinary dungeon/arena mobs.
  // When the player walks behind their ground shadow, their sprite body is redrawn on top.
  ppaDrawWorldBodyOccluders();

  // V89 depth sorting for the large clan boss.
  // If the player's feet are behind the boss, the boss body covers the player.
  if(P.scene==='clanboss1'){
    const _depthBoss=clanBossCombatTarget();
    if(_depthBoss)drawClanBossBodyOccluder(_depthBoss);
  }

  // World boss depth: player disappears behind the Titan when walking behind its body.
  if(P.scene==='worldboss'){
    const _worldDepthBoss=worldCrystalBossEntity();
    if(_worldDepthBoss)drawWorldCrystalBossOccluder(_worldDepthBoss);
  }

  // V204: small frameless character name follows the hero in world space.
  try{drawPlayerNickname()}catch(_){}

  drawMinimap();

  // Return to physical screen coordinates for screen-space effects.
  cx.restore();

  // Vignette — cached until the canvas size changes.
  cx.fillStyle=ppaGetVignetteGradient();cx.fillRect(0,0,cv.width,cv.height);
}

function drawMinimap(){
  const mm=document.getElementById('mmap');
  if(P.scene!=='dungeon'){
    __v218MinimapAt=0;
    if(mm.style.display!=='none')mm.style.display='none';
    return;
  }
  mm.style.display='block';
  const now=(typeof performance!=='undefined'&&performance.now)?performance.now():Date.now();
  if(now-__v218MinimapAt<320)return;
  __v218MinimapAt=now;
  const m=mm.getContext('2d');

  m.clearRect(0,0,200,133);

  }

  const groundDrop=12; // visually seat every NPC on the town floor
  for(const npc of ((SCENES[P.scene]&&SCENES[P.scene].npcs)||[])){
    if(npc.hidden)continue;
    const nx=npc.x-cam.x,ny=npc.y-cam.y;
    if(nx<-220||ny<-260||nx>cv.width+220||ny>cv.height+220)continue;
    const gy=ny+groundDrop;
    const glow=Math.sin(t*3+npc.x*0.001)*0.12+0.88;

    // Dark contact shadow so the NPC base visibly touches the ground.
    cx.fillStyle='rgba(0,0,0,0.30)';

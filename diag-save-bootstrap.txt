function pvpRoundResult(result){
  if(!PVP_MATCH.active)return {ok:false,error:'no_match'};
  pvpCombatEnabled=false;pvpWarmupUntil=0;pvpFightBannerUntil=0;
  let winner=null;
  if(result==='blue'||result==='red')winner=result;
  else if(result==='win')winner=PVP_LOCAL_SIDE;
  else if(result==='loss')winner=PVP_LOCAL_SIDE==='blue'?'red':'blue';
  if(!winner)return {ok:false,error:'bad_result'};
  PVP_MATCH[winner]++;
  const localWonRound=winner===PVP_LOCAL_SIDE;
  pvpRoundBannerText=localWonRound?'РАУНД ВЫИГРАН':'РАУНД ПРОИГРАН';
  pvpRoundBannerUntil=Date.now()+1800;
  if(PVP_MATCH.blue>=PVP_MATCH.firstTo||PVP_MATCH.red>=PVP_MATCH.firstTo){
    const localWon=PVP_MATCH[PVP_LOCAL_SIDE]>=PVP_MATCH.firstTo;
    setTimeout(function(){pvpFinishMatch(localWon)},850);
    return {ok:true,finished:true,score:pvpMatchScoreText()}
  }
  PVP_MATCH.round++;
  setTimeout(function(){pvpStartNextRound()},1800);
  return {ok:true,finished:false,round:PVP_MATCH.round,score:pvpMatchScoreText()}
}
function pvpStartConfirmedMatch(info){
  info=info||{};
  let mode=String(info.mode||'1x1').toLowerCase().replace('×','x');
  if(mode==='pvp11'||mode==='11')mode='1x1';
  if(mode==='pvp33'||mode==='33')mode='3x3';
  if(mode==='pvp55'||mode==='55')mode='5x5';
  if(!['1x1','3x3','5x5'].includes(mode))return {ok:false,error:'bad_mode'};
  if(PVP_MATCH.active)return {ok:false,error:'match_active'};
  if(!pvpConsumeAttempt(mode))return {ok:false,error:'no_attempts'};
  const spec=pvpMatchSpec(mode);
  PVP_PREVIEW_MODE=mode;
  PVP_LOCAL_SIDE=info.side==='red'?'red':'blue';
  PVP_MATCH={active:true,mode:mode,side:PVP_LOCAL_SIDE,matchId:info.matchId||null,round:1,blue:0,red:0,firstTo:spec.firstTo,maxRounds:spec.maxRounds,attemptSpent:true,settled:false,lastSettlement:null,stats:pvpFreshStats(),lastRecord:null};
  const ov=document.getElementById('pvpResultOverlay');if(ov)ov.style.display='none';
  closeArenaMenu();
  changeScene(spec.scene);
  return {ok:true,mode:mode,attemptsLeft:pvpAttemptLeft(pvpAttemptKind(mode)),firstTo:spec.firstTo,maxRounds:spec.maxRounds}
}

  return m.walk[gy*m.gw+gx]===1
}
function pvpCircleWalk(scene,x,y,r){
  const d=r*.78;
  return pvpPointWalk(scene,x,y)&&pvpPointWalk(scene,x+d,y)&&pvpPointWalk(scene,x-d,y)&&pvpPointWalk(scene,x,y+d)&&pvpPointWalk(scene,x,y-d)&&pvpPointWalk(scene,x+d*.7,y+d*.7)&&pvpPointWalk(scene,x-d*.7,y+d*.7)&&pvpPointWalk(scene,x+d*.7,y-d*.7)&&pvpPointWalk(scene,x-d*.7,y-d*.7)
}
function pvpSlide(scene,ox,oy,dx,dy,r){
  let nx=ox+dx,ny=oy+dy;
  if(pvpCircleWalk(scene,nx,ny,r))return {x:nx,y:ny};
  if(pvpCircleWalk(scene,nx,oy,r))return {x:nx,y:oy};
  if(pvpCircleWalk(scene,ox,ny,r))return {x:ox,y:ny};
  return {x:ox,y:oy}
}

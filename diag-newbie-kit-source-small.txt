function getDirRow(){
  // During standing attack OR run-and-attack, face the target/shot while
  // movement itself remains controlled by the joystick.
  if((P.attacking||P.runAttackT>0||P.shootT>0) && P.meleeAng !== undefined){
    const ax = Math.cos(P.meleeAng);
    const ay = Math.sin(P.meleeAng);
    if(Math.abs(ay) > Math.abs(ax) * 0.8){
      return ay > 0 ? 1 : 0;
    }
    return 2;
  }
  // During movement, use velocity with smooth direction switching
  const sp = Math.sqrt(P.vx*P.vx + P.vy*P.vy);
  if(sp > 0.5){
    // Use a threshold that favors vertical for more natural 4-dir movement
    if(Math.abs(P.vy) > Math.abs(P.vx) * 0.8){
      P.lastDir = P.vy > 0 ? 1 : 0;
    } else if(Math.abs(P.vx) > Math.abs(P.vy) * 0.8) {
      P.lastDir = 2;
    } else {
      // Diagonal: pick based on dominant axis
      if(Math.abs(P.vy) >= Math.abs(P.vx)) {
        P.lastDir = P.vy > 0 ? 1 : 0;
      } else {
        P.lastDir = 2;
      }
    }
  }
  return P.lastDir;
}

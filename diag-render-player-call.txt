function drawPlayer(){
  const sx=P.x-cam.x,sy=P.y-cam.y;
  const siegeVisualScale=P.scene==='clansiege'?CLAN_SIEGE_PLAYER_VISUAL_SCALE:1;

  // V277: only the player's 8 classes are drawn 10 px smaller.
  // No collision, camera, NPC, mob, AI or combat values are changed.
  const PLAYER_CLASS_EXTRA_REDUCTION_PX=10;
  const playerClassDrawHeight=(baseHeight)=>{
    const sc=Math.max(.01,Number(siegeVisualScale)||1);
    return Math.max(8,phoneCharacterDrawHeight(baseHeight,siegeVisualScale)-PLAYER_CLASS_EXTRA_REDUCTION_PX/sc);
  };
  const playerClassBodySize=(baseSize)=>{
    const sc=Math.max(.01,Number(siegeVisualScale)||1);
    return Math.max(8,phoneCharacterBodySize(baseSize,siegeVisualScale)-PLAYER_CLASS_EXTRA_REDUCTION_PX/sc);
  };
  const isGnome=playerUsesGnomeSprites();
  const isArcher=playerUsesArcherSprites();
  const isAssassinCustom=playerUsesAssassinSprites()&&!!ASSASSIN_ANIM[P.anim];
  const isTank=playerUsesTankSprites()&&!!TANK_ANIM[P.anim];
  const isBerserker=playerUsesBerserkerSprites()&&!!BERSERKER_ANIM[P.anim];
  const isPriest=playerUsesPriestSprites()&&!!PRIEST_ANIM[P.anim];
  const isMage=playerUsesMageSprites()&&!!MAGE_ANIM[P.anim];
  const isPaladin=playerUsesPaladinSprites()&&!!PALADIN_ANIM[P.anim];
  const isCustomDirectional=isGnome||isArcher||isAssassinCustom||isTank||isBerserker||isPriest||isMage||isPaladin;
  const phoneBodyBase=playerClassBodySize(P.sz);
  const visualBody=(isGnome?phoneBodyBase*0.72:phoneBodyBase)*siegeVisualScale;
  const bob=P.scene==='fartzone'?0:Math.sin(P.bob)*(isGnome?2.0:3);

  // Shadow follows the visible character size.
  // Fart Zone: exact feet-to-ground anchor, no visual hovering.
  const playerShadowY=P.scene==='fartzone'?sy+visualBody*0.40:sy+visualBody*0.45;
  cx.fillStyle='rgba(0,0,0,0.4)';
  cx.beginPath();cx.ellipse(sx,playerShadowY,visualBody*0.38,visualBody*0.14,0,0,Math.PI*2);cx.fill();

  // Dust when running
  if((Math.abs(P.vx)>0.5||Math.abs(P.vy)>0.5)&&Math.random()<0.25)
    PT.push({x:P.x+(Math.random()-.5)*(isGnome?6:12)*siegeVisualScale,y:P.y+visualBody*0.35,
      vx:-P.vx*0.4+(Math.random()-.5),vy:-P.vy*0.2+(Math.random()-.5)*0.5,
      life:10,ml:10,sz:(isGnome?1.5:2)*siegeVisualScale,col:'rgba(120,100,70,0.4)'});

  cx.save();
  cx.translate(sx,sy+bob);
  if(P.scene==='clansiege')cx.scale(CLAN_SIEGE_PLAYER_VISUAL_SCALE,CLAN_SIEGE_PLAYER_VISUAL_SCALE);

  // Legacy lean/recoil is disabled for the dwarf because his approved
  // 4-frame attack already contains the physical movement.
  if(!isCustomDirectional){

let __ppaVignetteGradient=null,__ppaVignetteKey='';
function ppaGetVignetteGradient(){
  const key=cv.width+'x'+cv.height;
  if(!__ppaVignetteGradient||__ppaVignetteKey!==key){
    __ppaVignetteKey=key;
    const g=cx.createRadialGradient(cv.width/2,cv.height/2,cv.height/3,cv.width/2,cv.height/2,cv.height);
    g.addColorStop(0,'rgba(0,0,0,0)');g.addColorStop(1,'rgba(0,0,0,0.65)');
    __ppaVignetteGradient=g;
  }
  return __ppaVignetteGradient;
}
// Camera zoom is scene-specific. Dungeon is brought a little closer while
// the town and arena keep their existing framing.
const CAMERA_ZOOM={safe:1,arena:1,dungeon:1.18,clanboss1:1.05,clansiege:1.38};
function cameraZoom(){return CAMERA_ZOOM[P.scene]||1}

function render(){
  const scCam=SCENES[P.scene];
  const mw=scCam.w||WORLD,mh=scCam.h||WORLD;
  const zoom=cameraZoom();
  const viewW=cv.width/zoom,viewH=cv.height/zoom;
  cam.x=P.x-viewW/2;cam.y=P.y-viewH/2;
  // Clamp camera to the map — no empty space beyond the edges
  cam.x=mw<=viewW?(mw-viewW)/2:Math.max(0,Math.min(cam.x,mw-viewW));
  cam.y=mh<=viewH?(mh-viewH)/2:Math.max(0,Math.min(cam.y,mh-viewH));

  // Clear/fill the physical screen first, then scale only the world.
  cx.setTransform(1,0,0,1,0,0);
  cx.fillStyle=P.scene==='safe'?'#3a3528':'#1a1a1a';
  cx.fillRect(0,0,cv.width,cv.height);
  cx.save();
  cx.scale(zoom,zoom);

  // Base fill: tiles for safe zone (no black void), dark for arena/dungeon
  if(P.scene==='safe'){
    const ts=96;
    for(let y=0;y<viewH;y+=ts){
      for(let x=0;x<viewW;x+=ts){
        const wx=x+cam.x,wy=y+cam.y;
        cx.fillStyle=((Math.floor(wx/ts)+Math.floor(wy/ts))%2===0)?'#3a3528':'#423d30';
        cx.fillRect(x,y,ts,ts);
      }
    }
  }else{
    cx.fillStyle='#1a1a1a';cx.fillRect(0,0,viewW,viewH);
  }

  // Background image — each scene is a single self-contained 1024x1024 map,
  // drawn exactly once at its world position. No tiling/repeating, so the two
  // locations never visually overlap or duplicate.
  const sc2=SCENES[P.scene];
  const bg=sc2.bg;
  if(bg.complete&&bg.naturalWidth>0){
    cx.drawImage(bg,-cam.x,-cam.y,sc2.w||WORLD,sc2.h||WORLD);
    if(P.scene==='clanboss1'&&CLAN_BOSS_SHOW_MASK&&imgClanBoss1Mask.complete&&imgClanBoss1Mask.naturalWidth>0){
      cx.save();cx.globalAlpha=.72;
      cx.drawImage(imgClanBoss1Mask,-cam.x,-cam.y,CLAN_BOSS_WORLD,CLAN_BOSS_WORLD);
      cx.restore();
    }
  }else{
    cx.fillStyle=P.scene==='safe'?'#1a1510':'#1a0a0a';
    cx.fillRect(0,0,cv.width,cv.height);
  }

  // Darken background slightly for readability
  cx.fillStyle='rgba(0,0,0,0.16)';
  cx.fillRect(0,0,viewW,viewH);

  // Town fence stays town-only. NPCs may exist in other peaceful scenes too.
  if(P.scene==='safe')drawFence();
  if(P.scene==='fartzone')drawFartZoneMines();
  if(SCENES[P.scene]&&SCENES[P.scene].npcs)drawNPC();
  if(P.scene==='clansiege')drawClanSiegeWorld();
  // V187: dungeon exit is a physical portal in the respawn room.
  if(sc2.portal)drawPortal();

  // Target marker (red circle)
  if(P.tid!=null){
    const t=EN.find(e=>e.id==P.tid);
    if(t&&targetIsValid(t)&&!(P.scene==='clanboss1'&&t.isClanBoss)){
      const tx=t.x-cam.x,ty=t.y-cam.y;
      const pulse=Math.sin(Date.now()*0.007)*0.15+0.85;
      // Semi-transparent red fill
      const targetR=(t.type&&t.type.slimeScavenger)?15:t.sz*1.3;
      cx.fillStyle='rgba(255,0,0,0.18)';
      cx.beginPath();cx.arc(tx,ty,targetR,0,Math.PI*2);cx.fill();
      // Red ring
      cx.strokeStyle='rgba(255,50,50,0.7)';cx.lineWidth=3;
      cx.beginPath();cx.arc(tx,ty,targetR*pulse,0,Math.PI*2);cx.stroke();
      // Rotating dots
      const ra=Date.now()*0.003;
      for(let i=0;i<4;i++){
        const a=ra+i*Math.PI/2;
        cx.fillStyle='rgba(255,90,90,0.85)';
        cx.beginPath();cx.arc(tx+Math.cos(a)*t.sz*1.5,ty+Math.sin(a)*t.sz*1.5,3,0,Math.PI*2);cx.fill();
      }
      // Target line from player
      cx.strokeStyle='rgba(255,50,50,0.2)';cx.lineWidth=1;
      cx.setLineDash([5,5]);
      cx.beginPath();cx.moveTo(P.x-cam.x,P.y-cam.y);cx.lineTo(tx,ty);cx.stroke();
      cx.setLineDash([]);
    }
  }

  // Particles (under entities)
  for(const p of PT){
    const px=p.x-cam.x,py=p.y-cam.y;
    if(px<-40||py<-40||px>viewW+40||py>viewH+40)continue;
    const a=p.life/p.ml,sz=Math.max(1,p.sz*a);
    cx.globalAlpha=a;cx.fillStyle=p.col;
    cx.fillRect(Math.round(px-sz*.5),Math.round(py-sz*.5),Math.ceil(sz),Math.ceil(sz));
  }
  cx.globalAlpha=1;

  // V106 corpses + death traces stay under living entities and have no collision.
  ppaDrawDeathRemnants();

  // Personal world-boss chests.
  drawWorldBossPersonalChests();

  // Loot on the floor
  for(const l of LOOT){
    const lx=l.x-cam.x,ly=l.y-cam.y+Math.sin(l.bob)*3;
    if(lx<-50||ly<-50||lx>viewW+50||ly>viewH+50)continue;
    cx.fillStyle='rgba(0,0,0,0.4)';cx.beginPath();cx.ellipse(lx,ly+10,10,4,0,0,Math.PI*2);cx.fill();
    if(l.kind==='gold'){
      const g=cx.createRadialGradient(lx,ly,0,lx,ly,16);
      g.addColorStop(0,'rgba(255,220,90,0.55)');g.addColorStop(1,'rgba(255,180,0,0)');
      cx.fillStyle=g;cx.beginPath();cx.arc(lx,ly,16,0,Math.PI*2);cx.fill();
      cx.font='16px monospace';cx.textAlign='center';cx.fillText('🪙',lx,ly
